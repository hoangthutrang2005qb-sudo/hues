2. Các Ràng Buộc Bất Biến (Security Invariants)

Security 100%
1. Cơ chế Soulbound (_update hook):
Chỉ cho phép đúc mới (from == 0) và thiêu hủy (to == 0). Bất kỳ lệnh chuyển nhượng ví nào (transferFrom, safeTransferFrom) đều bị revert ngay lập tức on-chain.
2. Một Sinh viên — Một Ví duy nhất:
Ngăn chặn việc 1 sinh viên tạo nhiều ví để nhận trùng lặp hoặc mượn ví của sinh viên khác.
3. Quyền thu hồi kỷ luật (OnlyOwner):
Khi sinh viên vi phạm quy chế đào tạo, Hội đồng Đại học Huế kích hoạt hàm revokeBadge() để thiêu hủy on-chain, lưu lại số quyết định kỷ luật vĩnh viễn.

💻 Mã Nguồn Cốt Lõi: Khóa Chuyển Nhượng Soulbound (Solidity)

ProjectCore.sol 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

import "@openzeppelin/contracts/token/ERC721/ERC721.sol";
import "@openzeppelin/contracts/access/Ownable.sol";

error SoulboundTokenNonTransferable();
error WalletAlreadyLinked();
error DuplicateBadgeType();

/**
 * @title ProjectCore - Hệ Thống Văn Bằng Số Đại Học Huế
 * @notice Khóa chuyển nhượng bất biến (Soulbound Token) theo chuẩn ERC-721 v5
 */
contract ProjectCore is ERC721, Ownable {
    // Override _update hook trong OpenZeppelin v5 để biến token thành Soulbound
    function _update(address to, uint256 tokenId, address auth) internal override returns (address) {
        address from = _ownerOf(tokenId);

        // Chỉ cho phép Mint (from == address(0)) hoặc Burn (to == address(0))
        if (from != address(0) && to != address(0)) {
            revert SoulboundTokenNonTransferable();
        }

        return super._update(to, tokenId, auth);
    }
}
